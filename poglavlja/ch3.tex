\section{Apsolutna vrijednost realnog broja. Arhimedov aksiom}
Sigurno ste se prije u obrazovanju susreli s pojmovima otvorenih, zatvorenih, poluotvorenih i poluzatvorenih intervala, za koje sad dajemo preciznu definiciju.
\begin{definition}
Neka su zadani \(a, b\in \mathbb{R}\), \(a<b\). Definiramo sljedeće skupove:
\begin{multicols}{2}
\begin{itemize}
\item \(\langle a, b\rangle:=\{x\in \mathbb{R} : a<x<b\}\),
\item \([a, b]:=\{x\in \mathbb{R} : a\leq x\leq b\}\),
\item \([a, b\rangle:=\{x\in \mathbb{R} : a\leq x< b\}\),
\item \(\langle a, b]:=\{x\in \mathbb{R} : a< x\leq b\}\),
\item \(\langle -\infty, b]:=\{x\in \mathbb{R} : x\leq b\}\),
\item \(\langle -\infty, b\rangle:=\{x\in \mathbb{R} : x< b\}\),
\item \([a, \infty\rangle:=\{x\in \mathbb{R} : a\leq x\}\),
\item \(\langle a, \infty\rangle:=\{x\in \mathbb{R} : a< x\}\).
\end{itemize}
\end{multicols}
Skupovi \(\langle a,b \rangle\), \(\langle a,\infty \rangle\), \(\langle -\infty,b \rangle\) su \textbf{otvoreni intervali}, skup \([a, b]\) je \textbf{zatvoreni interval}, a skupovi \([a, b\rangle\), \(\langle a, b]\), \(\langle -\infty, b]\), \([a, \infty\rangle\) su \textbf{poluotvoreni intervali}.
\end{definition}
Još jedan već otprije poznat pojam je apsolutna vrijednost realnog broja.
\begin{definition}
\textbf{Apsolutna vrijednost} broja $x\in \mathbb{R}$ je broj
$$\left\lvert x\right\rvert:=\begin{cases}
x, & x\geq 0,\\
-x & x<0.
\end{cases}$$
\end{definition}
\begin{remark}[Nejednakost trokuta]
\label{triangleineq}
Za sve $x, y\in \mathbb{R}$ vrijedi $\lvert x+y\rvert\leq \lvert x\rvert + \lvert y\rvert$.
\end{remark}
\begin{remark}[O apsolutnoj vrijednosti razlike]
\label{invtriangleineq}
Za sve $x, y\in \mathbb{R}$ vrijedi $\lvert\lvert x\rvert-\lvert y\rvert\rvert\leq \lvert x-y\rvert$.
\end{remark}
\begin{remark}[Apsolutna vrijednost produkta i kvocijenta] \textbf{}
\begin{itemize}
\item[a)] Za sve $x, y\in \mathbb{R}$ vrijedi $\lvert xy\rvert=\lvert x\rvert\lvert y\rvert$. 
\item[b)] Za sve $x\in \mathbb{R}$ i $y\in \mathbb{R}\setminus\{0\}$ vrijedi $\left\lvert \dfrac{x}{y}\right\rvert=\dfrac{\lvert x\rvert}{\lvert y\rvert}$.
\end{itemize}
\end{remark}
\begin{remark}
\label{absolutevaluelemma}
Za sve $x\in \mathbb{R}$ i za sve $\epsilon\geq 0$ vrijedi
$$\lvert x\rvert\leq \epsilon\Leftrightarrow -\epsilon\leq x\leq \epsilon.$$
Ako je $\epsilon>0$, analogna tvrdnja vrijedi u slučaju stroge nejednakosti.
\end{remark}
\begin{exercise} \textbf{}
\begin{itemize}
\item[a)] Odredite sve $x\in \mathbb{R}$ takve da je $\left\lvert x+2\right\rvert+\left\lvert x+3\right\rvert=3$.
\item[b)] Odredite sve $x\in \mathbb{R}$ takve da je $\lvert x+3\rvert\leq 7$ i $\lvert x-2\rvert< 3-x$.
\end{itemize}
\end{exercise}
\begin{proof}[Rješenje]
a) Neka je $x\in \mathbb{R}$ proizvoljan. Ako je $x+2\geq 0$, onda je i $x+3\geq 0$. pa je početna jednadžba ekvivalentna s $$x+2+x+3=3,$$ tj. dobivamo rješenje $x_1=-1$, i zaista vrijedi $x_1+2\geq 0$. Ako je pak $x+3\geq 0$ i $x+2<0$ onda imamo $$-x-2+x+3=3,$$ tj. $1=3$, što je nemoguće. Ako je $x+3< 0$, onda je i $x+2< 0$, pa imamo $$-x-3-x-3=3,$$ tj. $x_2=-4$, te i ovo rješenje zadovoljava $x_2+3<0$. Stoga su jedini brojevi koji zadovoljavaju početnu jednadžbu $-1$ i $-4$.

b) Neka je $x\in \mathbb{R}$ proizvoljan. Tvrdnja $\lvert x+3\rvert\leq 7$ je ekvivalentna s $-7\leq x+3\leq 7$, tj. $-10\leq x\leq 4$. Ako je $x> 3$, $\lvert x-2\rvert< 3-x$ očigledno ne vrijedi jer je $\lvert x-2\rvert\geq 0$ i $3-x<0$. Ako je $x=3$ imamo $1<0$, što također ne vrijedi, a ako je $x<3$, onda je $3-x>0$ i tvrdnja je ekvivalentna s $$-3+x<x-2<3-x,$$ pa dobivamo $-3<-2$ i $x< \dfrac{5}{2}$ i očito je $x<3$. Dakle, tvrdnje iz zadatka vrijede ako i samo ako je $-10\leq x<\dfrac{5}{2}$, odnosno $x\in \left[ -10, \dfrac{5}{2}\right\rangle$.
\end{proof}
\begin{exercise} \textbf{}
\begin{itemize}
\item[a)] Neka su $a_1, a_2\in \mathbb{R}$ takvi da je $\left\lvert a_1-a_2\right\rvert\leq 1$. Dokažite da je $\left\lvert a_1\right\rvert\leq 1+\left\lvert a_2\right\rvert$.
\item[b)] Neka su $a, a', b, b'\in \mathbb{R}$ takvi da je $\lvert a-a'\rvert\leq 1$ i $\lvert b-b'\rvert\leq 1$. Dokažite da je $\lvert ab'- a'b\rvert\leq \lvert a'\rvert+\lvert b'\rvert$.
\end{itemize}
\end{exercise}
\begin{proof}
a) Vrijedi
$$\left\lvert a_1\right\rvert=\left\lvert \left(a_1-a_2\right)+a_2\right\rvert\leq \left\lvert a_1-a_2\right\rvert + \left\lvert a_2\right\rvert\leq 1+ \left\lvert a_2\right\rvert.$$

b) Želimo da se "pojave" izrazi $\lvert a-a'\rvert$ i $\lvert b-b'\rvert$, pa je ideja ovdje dodati i oduzeti $a'b'$. Vrijedi
\begin{align*}
\lvert ab'-a'b\rvert&=\lvert \left(ab'-a'b'\right)+\left(a'b'-a'b\right)\rvert\leq \lvert ab'-a'b'\rvert+\lvert a'b'-a'b\rvert\\
&\leq \lvert b'\left(a-a'\right)\rvert+\lvert a'\left(b'-b\right)\rvert=\lvert b'\rvert\lvert a-a'\rvert+\lvert a'\rvert\lvert b-b'\rvert\leq \lvert a'\rvert+\lvert b'\rvert.
\end{align*}
Ovdje smo iskoristili i činjenicu da je $\lvert -x\rvert=\lvert x\rvert$, za svaki $x\in \mathbb{R}$. Naime, neka je $x\in \mathbb{R}$. Ako je $x\geq 0$ imamo $-x\leq 0$, pa je $\lvert -x\rvert=-(-x)=x=\lvert x\rvert$, a ako je $x\leq 0$, onda je $-x\geq 0$, pa je $\lvert -x\rvert=-x=\lvert x\rvert$.
\end{proof}
\begin{remark}[Aksiomi realnih brojeva]
\label{axioms}
Aksiomi realnih brojeva su sljedeća svojstva skupa $\mathbb{R}$.
\begin{itemize}
\item Asocijativnost zbrajanja -- Za sve $x, y, z\in \mathbb{R}$ vrijedi $(x+y)+z=x+(y+z)$.
\item Postojanje neutralnog elementa (nule) za "$+$" -- Postoji jedinstveni element $0\in \mathbb{R}$ takav da vrijedi $x+0=x$ za sve $x\in \mathbb{R}$.
\item Postojanje inverznog (suprotnog) elementa za "$+$" -- Za svaki $x\in \mathbb{R}$ postoji jedinstveni element $-x\in \mathbb{R}$ takav da je $x+(-x)=(-x)+x=0$.
\item Komutativnost za "$+$" -- Za sve $x, y\in \mathbb{R}$ vrijedi $x+y=y+x$.
\item Asocijativnost za "$\cdot$" -- Za sve $x, y, z\in \mathbb{R}$ vrijedi $(x\cdot y)\cdot z=x\cdot (y\cdot z)$.
\item Postojanje neutralnog elementa (jedinice) za "$\cdot$" -- Postoji jedinstveni element $1\neq 0$ takav da vrijedi $x\cdot 1=x$ za sve $x\in \mathbb{R}$.
\item Postojanje inverznog (recipročnog) elementa za "$\cdot$" -- Za svaki $x\in \mathbb{R}\setminus \{0\}$ postoji jedinstveni element $x^{-1}=\dfrac{1}{x}$ takav da je $x\cdot \dfrac{1}{x}=\dfrac{1}{x}\cdot x=1$. 
\item Komutativnost za "$\cdot$" -- Za sve $x, y\in \mathbb{R}$ vrijedi $x\cdot y=y\cdot x$.
\item Distributivnost $\cdot$ nad "$+$" -- Za sve $x, y, z\in \mathbb{R}$ vrijedi $x\cdot (y+z)=x\cdot y+x\cdot z$.
\item Trihotomija -- Za sve \(x, y\in \mathbb{R}\) vrijedi $x\leq y$ ili $y\leq x$.
\item Antisimetričnost -- Za sve \(x, y\in \mathbb{R}\) vrijedi
$x\leq y\; \wedge\; y\leq x\Rightarrow x=y$.
\item Tranzitivnost -- Za sve \(x, y, z\in \mathbb{R}\) vrijedi
\[x\leq y\; \wedge\; y\leq z\Rightarrow x\leq z,\hspace{0.3cm}x< y\; \wedge\; y< z\Rightarrow x< z,\]
\item Kompatibilnost s "\(+\)" -- Za sve \(x, y, z\in \mathbb{R}\) vrijedi
\[x\leq y\Leftrightarrow x+z\leq y+z,\hspace{0.3cm}x< y\Leftrightarrow x+z< y+z,\]
\item Kompatibilnost s "\(\cdot\)" -- Za sve \(x, y, z\in \mathbb{R}\) vrijedi
\[x\leq y\wedge z>0\Leftrightarrow xz\leq yz,\hspace{0.3cm}x< y\wedge z>0\Leftrightarrow xz< yz.\]
\item Aksiom potpunosti -- Svaki odozgo ograničen skup ima supremum.
\end{itemize}
\end{remark}

Ovi aksiomi se možda čine očiglednima (i jesu), ali njihova teorijska važnost u tome je da oni, intuitivno govoreći, potpuno opisuju skup $\mathbb{R}$ do na "promjenu oznaka", tj. ako neki drugi skup ima gornja svojstva, onda je taj skup u određenom smislu zapravo samo skup $\mathbb{R}$ s eventualno možda drugačijim oznakama za brojeve. Aksiom potpunosti ćemo još prokomentirati kasnije kada budemo uveli pojam supremuma.

Još jedna bitna činjenica je da se (gotovo) svaki teorem o realnim brojevima može dokazati iz aksioma realnih brojeva. Tako se npr. sljedeći važan teorem može primjenom tih aksioma relativno lako dokazati. 
\begin{remark}[Arhimedov aksiom]
Neka su \(a>0\) i \(b>0\) proizvoljni. Tada postoji \(n\in \mathbb{N}\) takav da je \(na>b\).
\end{remark}
Ovaj teorem, koji intuitivno govori da se za dužinu bilo koje duljine može konstruirati od nje dulja dužina uzastopnim nanošenjem dužine bilo koje duljine, je neophodan u analizi, pa u nastavku dajemo zadatke vezane uz primjenu ovog teorema.
\begin{exercise} Dokažite da za svaki realan \(r>0\)...
\begin{itemize}
\item[a)] ...postoji bar jedan \(a\in \mathbb{N}\) takav da je \(\dfrac{1}{a}<r\). Postoji li i beskonačno mnogo prirodnih brojeva koji zadovoljavaju tvrdnju?
\item[b)] ...postoji bar jedan \(a\in \mathbb{N}\) takav da je \(\dfrac{2}{a^2+1}<r\).
\item[c)] ...postoji bar jedan \(a\in \mathbb{N}\) takav da je \(\dfrac{1}{5^a+1}<r\).
\end{itemize}
\end{exercise}
\begin{proof*}
a) Prema Arhimedovu aksiomu, za svaki \(r>0\) postoji \(a\in \mathbb{N}\) takav da je \(ra>1\), odnosno \(\dfrac{1}{a}<r\). Nadalje, za svaki \(b\in \mathbb{N}\) takav da je \(b>a\) vrijedi \[\dfrac{1}{b}<\dfrac{1}{a}<r.\]
Dakle, postoji beskonačno mnogo brojeva $b$ za koje vrijedi tvrdnja.

b) Prema Arhimedovu aksiomu, za svaki \(r>0\) postoji \(a\in \mathbb{N}\) takav da je \(ra>2\). Odavde imamo
$$r(a^2+1)>ra>2\Rightarrow \dfrac{2}{a^2+1}<r.$$

c) Prema Arhimedovu aksiomu, za svaki \(r>0\) postoji \(a\in \mathbb{N}\) takav da je \(ra>1\). Dokažimo matematičkom indukcijom da je $5^n+1>n$ za sve $n\in \mathbb{N}$. Za $n=1$ tvrdnja vrijedi, a pretpostavimo li da tvrdnja vrijedi za neki $n$, imamo
$$5^{n+1}+1=5^n+1+4\cdot 5^n>n+4\cdot 5^n\geq n+1,$$
što daje tvrdnju. Dobivamo
\[\pushQED{\qed}
r(5^a+1)>ra>1\Rightarrow \dfrac{1}{5^a+1}<r. \qedhere
\popQED
\]
\end{proof*}
\begin{exercise}
\label{archimedes2}
Dokažite da za svaki $\epsilon>0$ postoji $r_0\in \mathbb{Q}$ takav da je $0<1-r_0^2<\epsilon$.
\end{exercise}
\begin{proof}[Rješenje]
Zapravo se traži racionalan broj $r_0$ takav da je $r_0^2<1$ i da je $r_0^2$ "dovoljno blizu" broju $1$. Ovdje je smisleno promatrati brojeve oblika $\dfrac{n-1}{n}$, jer je od svih racionalnih brojeva između $0$ i $1$ čiji je nazivnik $n$ kvadrat brojeva tog oblika "najbliži" broju $1$. Općenito, ako je $q=\dfrac{n-1}{n}$, gdje je $n\in \mathbb{N}$, onda je 
$$1-\dfrac{(n-1)^2}{n^2}=\dfrac{n^2-(n-1)^2}{n^2}=\dfrac{2n-1}{n^2}.$$
Vrijedi $$\dfrac{2n-1}{n^2}<\epsilon\Leftrightarrow \dfrac{n^2}{2n-1}\epsilon>1\Leftrightarrow \dfrac{n^2-\dfrac{1}{2}n+\dfrac{1}{2}n}{2n-1}\epsilon>1\Leftrightarrow \left(\dfrac{1}{2}n+\dfrac{n}{2(2n-1)}\right)\epsilon>1.$$
U tu svrhu, uzmimo $n_0\in \mathbb{N}$ takav da je $n_0\cdot\dfrac{\epsilon}{2}>1$ i definirajmo $r_0:=\dfrac{n_0-1}{n_0}$. Tada je
$$1<n_0\dfrac{\epsilon}{2}<\left(\dfrac{1}{2}n_0+\dfrac{n_0}{2(2n_0-1)}\right)\epsilon,$$
što skupa s činjenicom da je $r_0<1$, odakle slijedi $1-r_0^2>0$ daje traženu tvrdnju.
\end{proof}
\begin{exercise}
\label{lemmaforalimitpointexercise}
Neka je $a\in \langle 0, 1\rangle$. Dokažite da postoji $m\in \mathbb{N}$ takav da za svaki $x\in \left\langle 0, \dfrac{1}{m}\right\rangle$ postoji $n\in \mathbb{N}$ takav da je $nx\in \langle a, 1\rangle$.
\end{exercise}
\begin{proof*}
Pretpostavimo suprotno, da za svaki $m\in \mathbb{N}$ postoji $x\in \left\langle 0, \dfrac{1}{m}\right\rangle$ takav da za svaki $n\in \mathbb{N}$ vrijedi $nx\notin \langle a, 1\rangle$. Po Arhimedovu aksiomu postoje $m_0, m_1\in \mathbb{N}$ takav da je $\dfrac{1}{m_0}<a$ i $\dfrac{1}{m_1}<1-a$. Uzmimo $m_2:=\max{\{m_0, m_1\}}$. Tada postoji $x_0\in \left\langle 0, \dfrac{1}{m_2}\right\rangle$ takav da za svaki $n\in \mathbb{N}$ vrijedi $nx_0\notin \langle a, 1\rangle$, te vrijedi
$$x_0<\dfrac{1}{m_2}\leq \dfrac{1}{m_0}<a.$$
Uzmimo najveći $n_0\in \mathbb{N}$ takav da je $n_0x_0\leq a$ (takvi $n$-ovi postoje, npr. $n=1$). Tada je $(n_0+1)x_0>1$, pa je $$(n_0+1)x_0-n_0x_0\geq 1-a,$$ tj. $x_0\geq 1-a$, što daje kontradikciju s činjenicom da je
\[\pushQED{\qed}
x_0<\dfrac{1}{m_2}\leq\dfrac{1}{m_1}<1-a. \qedhere
\popQED
\]
\end{proof*}
\begin{remark}[Gustoća \(\mathbb{Q}\) i \(\mathbb{I}\) u \(\mathbb{R}\)] Vrijedi sljedeće.
\label{den}
\textbf{}
\begin{itemize}
\item[a)] Za svaki \(\epsilon>0\) i za sve \(x\in \mathbb{R}\), \(\langle x-\epsilon, x+\epsilon\rangle\cap \mathbb{Q}\neq \emptyset\),
\item[b)] Za svaki \(\epsilon>0\) i za sve \(x\in \mathbb{R}\), \(\langle x-\epsilon, x+\epsilon\rangle\cap \mathbb{I}\neq \emptyset\).
\end{itemize}
\end{remark}
Uočimo da je napomena \ref{den} ekvivalentna tvrdnji da za sve \(c<d\), \(c, d\in \mathbb{R}\) postoji \(x\in \mathbb{Q}\) (odnosno $x\in \mathbb{I}$) takav da je \(c<x<d\). Zaista, uvrstimo li u napomenu \ref{den} \(x:=\dfrac{c+d}{2}\), \(\epsilon:=\dfrac{d-c}{2}\), dobivamo da postoji bar jedan \(x\in \mathbb{Q}\) (odnosno \(x\in \mathbb{I}\)) u intervalu \(\langle c, d\rangle\), tj. bar jedan \(x\in \mathbb{Q}\) (odnosno \(x\in \mathbb{I}\)) takav da je \(c<x<d\), što smo i tvrdili. Obratna implikacija se vidi za $c:=x-\epsilon$ i $d:=x+\epsilon$. Drugim riječima, između svaka dva realna broja postoji bar jedan racionalan broj i bar jedan iracionalan broj. Ovo svojstvo se također naziva \textit{gustoća skupova} $\mathbb{Q}$ \textit{i} $\mathbb{I}$ \textit{u} $\mathbb{R}$.

Napomena \ref{den} je često korisna ako treba dokazati egzistenciju nekog racionalnog ili iracionalnog broja u nekom intervalu bez da ga eksplicitno konstruiramo.
\begin{exercise} \textbf{}
\begin{itemize}
\item[a)] Riješite zadatak \ref{archimedes2} koristeći gustoću skupa $\mathbb{Q}$ u $\mathbb{R}$.
\item[b)] Dokažite tvrdnju iz b) dijela napomene \ref{den}. 
\end{itemize}
\end{exercise}
\begin{proof}[Rješenje]
a) Neka je $\epsilon>0$ proizvoljan. Ako je $\epsilon\geq 1$, onda za $r_0:=\dfrac{1}{2}$ vrijedi 
$$0<1-r_0^2=\dfrac{3}{4}<1\leq \epsilon.$$
Ako je $\epsilon<1$, onda po gustoći skupa $\mathbb{Q}$ u $\mathbb{R}$ postoji $r_0\in \left\langle \sqrt{1-\epsilon}, 1\right\rangle$. Tada je $r_0^2<1$, odnosno $0<1-r_0^2$ i vrijedi $r_0^2>1-\epsilon$, odnosno $1-r_0^2<\epsilon$.

b) Neka su \(x\in \mathbb{R}\) i \(\epsilon>0\) proizvoljni. Uočimo da ako je jedan od brojeva \(x-\epsilon\), \(x+\epsilon\) racionalan, a drugi iracionalan, onda je traženi broj \(\dfrac{x-\epsilon+x+\epsilon}{2}=x\), jer je aritmetička sredina racionalnog i iracionalnog broja opet iracionalan broj. Pretpostavimo sad da su oba broja iracionalna. Tada postoji \(q\in \mathbb{Q}\) takav da je \(x<q<x+\epsilon\), pa je \(\dfrac{q+x+\epsilon}{2}\) traženi broj.

Ako su pak oba broja racionalna, onda je i \(x+\epsilon-(x-\epsilon)=2\epsilon\) racionalan broj, pa je i \(\epsilon\) racionalan broj. Dakle, ako je \(x\in \mathbb{Q}\), onda je \(x+\dfrac{\epsilon}{\sqrt{2}}\) traženi broj (uočite da je \(\dfrac{\epsilon}{\sqrt{2}}\) iracionalan), a ako je \(x\in \mathbb{I}\), onda je \(x\) traženi broj.
\end{proof}
\begin{remark}
Za one upoznate s osnovnim konceptima teorije skupova, evo kako se još može riješiti c) dio prethodnog zadatka. Pretpostavimo suprotno, da postoje \(x\in \mathbb{R}\) i \(\epsilon>0\) takvi da je \(\langle x-\epsilon, x+\epsilon\rangle\cap \mathbb{I}=\emptyset\). To znači da je \(\langle x-\epsilon, x+\epsilon\rangle\subseteq \mathbb{Q}\), pa je \(\langle x-\epsilon, x+\epsilon\rangle\) konačan ili prebrojiv, budući da je \(\mathbb{Q}\) prebrojiv, te je podskup svakog prebrojivog skupa konačan ili prebrojiv. Međutim, znamo da su otvoreni skupovi neprebrojivi (jer su u bijekciji s \(\mathbb{R}\)), što daje kontradikciju.
\end{remark}
\section{Supremum i infimum}
\begin{definition}
\label{minimax}
Kažemo da je \(a\in \mathbb{R}\) \textbf{gornja međa} skupa \(S\subseteq{\mathbb{R}}\) ako za svaki \(x\in S\) vrijedi \(x\leq a\), odnosno \textbf{donja međa} skupa \(S\) ako za svaki \(x\in S\) vrijedi \(a\leq x\). Ako je pritom \(a\in S\), onda je \(a\) \textbf{maksimum} (ako je on gornja međa), odnosno \textbf{minimum} (ako je on donja međa) skupa \(S\). Maksimum skupa \(S\) označavamo sa \(\max{S}\), a minimum s \(\min{S}\). 

Ako skup ima gornju (donju među), onda kažemo da je odozgo (odozdo) ograničen/omeđen, u suprotnome je odozgo (odozdo) ograničen/omeđen. Ako ima obje, onda samo kažemo da je ograničen/omeđen, u suprotnome je neograničen/neomeđen.
\end{definition}

\begin{exmp}
Skup \(T=\{1, 2, 3, 4, 5.5, 8\}\) je odozgo ograničen, jer je \(8\) jedna njegova gornja međa, a \(8\) je i maksimum jer je \(8\in T\). Uočite da su njegove gornje međe npr. \(9\), \(12\) te \(2000\) (Općenito, svi brojevi iz skupa \([8,\infty\rangle\)). S druge strane, skup \(\mathbb{N}\) nije odozgo ograničen.
\end{exmp}
\begin{exercise} \textbf{}
\begin{itemize}
\item[a)] ...skupa \(A\subseteq \mathbb{R}\) koji je odozgo ograničen, odozdo neograničen, te nema maksimum.
\item[b)] ...beskonačnog skupa \(A\subseteq \mathbb{R}\) takvog da je \(\min{A}=0\), \(\max{A}=5\), te vrijedi \(1\notin A\) i \(3\notin A\).
\end{itemize}
\end{exercise}
\begin{proof}[Rješenje]
a) Jedan takav skup je npr. \(A:=\langle -\infty, 3\rangle\). Zaista, očito je odozgo ograničen (vrijedi \(x\in A\) ako i samo ako je \(x<3\), dakle jedna gornja međa je npr. \(4\)), odozdo neograničen (za svaki \(a\in \mathbb{R}\) postoji \(x\in A\) takav da je \(x<a\), npr. bilo koji član skupa \(A\) ako je \(a\geq 3\) i \(a-1\) ako je \(a<3\)), te nema maksimum (kad bi postojao maksimum \(b\), po definiciji je \(b<3\), no \(b<\dfrac{b+3}{2}<3\), što je kontradikcija s činjenicom da je \(b\) maksimum).

Još nekoliko primjera skupova s danim svojstvom: \(\mathbb{N}^-\cup [3, 4\rangle\), \(\langle -\infty, 2]\cap \mathbb{I}\).

b) Uzmimo npr. \(A:=\left[0,\dfrac{1}{2}\right]\cup \left[4, 5\right]\). Uvjerimo se npr. da je \(\min{A}=0\). Neka je \(x\in A\) proizvoljan. Vrijedi \(x\in A\) ako i samo ako vrijedi \(x\in \left[0,\dfrac{1}{2}\right]\) ili \(x\in [4, 5]\). Ako je \(x\in \left[0,\dfrac{1}{2}\right]\), tj. \(0\leq x\leq \dfrac{1}{2}\), onda je očito i \(x\geq 0\), isto vrijedi i ako je \(x\in [4, 5]\). Dakle, \(0\leq x\) za sve \(x\in A\). Kako je \(0\in A\), zaključujemo da je \(\min{A}=0\). Analogno se zaključuje da je \(\max{A}=5\). Očito vrijedi \(1\notin A\), jer bi inače trebalo vrijediti \(0\leq 1\leq \dfrac{1}{2}\) ili \(4\leq 1\leq 5\), a nikoja od te dvije tvrdnje nije istinita. Slično vidimo i da vrijedi \(3\notin A\).
\end{proof}
\begin{definition}
Neka je \(S\) odozgo ograničen skup. \textbf{Supremum} skupa \(S\) je najmanja gornja međa od \(S\).
\end{definition}

\begin{definition}
Neka je \(S\) odozdo ograničen skup. \textbf{Infimum} skupa \(S\) je najveća donja međa od \(S\).
\end{definition}

Kako su supremum i infimum jedinstveni ako postoje, ima smisla uvesti oznake \(\sup{A}\) i \(\inf{A}\).
Napomenimo da infimum (supremum) odozdo (odozgo) ograničenog skupa može, ali i ne mora biti unutar tog skupa. 
\begin{remark}
\label{supmax}
Vrijedi sljedeće -- ako je \(S\subseteq \mathbb{R}\) odozgo omeđen skup koji ima maksimum, onda je \(\sup{S}=\max{S}\). Zaista, kad bi postojao \(a\in S\) takav da je \(a<\max{S}\) i da za sve \(x\in S\) vrijedi \(x\leq a\), onda \(\max{S}\in S\) povlači \(\max{S}\leq a\), što je u kontradikciji s \(a<\max{S}\). Analogno se pokaže da ako je \(S\) odozdo omeđen skup koji ima minimum, onda je \(\inf{S}=\min{S}\).
\end{remark}

Zaista, pretpostavimo li da \(1\) nije supremum, onda kako je $1$ jedna gornja međa, postoji neka gornja međa \(M\) takva da je \(M<1\), dakle vrijedi \(a\leq M\) za svaki \(a\in [0, 1\rangle\). Očito je \(M\geq 0\), što dobivamo uvrštavanjem $a=0$. Sada dobivamo kontradikciju s činjenicom da je \[\dfrac{M+1}{2}>M\;\;\text{i}\;\;0\leq \dfrac{M+1}{2}<1,\] što znači da je \(\dfrac{M+1}{2}\in [0, 1\rangle\).

$$\left(-2\leq x\leq 2\right)\;\;\wedge\;\;\left(x<-1\vee x>-1\right).$$

odnosno $x\in S'$.

Očito je $\max{S}=\max{\left(S\setminus S'\right)}=\sup{S}=\sup{\left(S\setminus S'\right)}=2$ i $\min{S}=\max{\left(S\setminus S'\right)}=\sup{S}=\sup{\left(S\setminus S'\right)}=2$.

\end{proof*}

\begin{proof}[Rješenje]
Za sve \(x\in \mathbb{R}\) vrijedi \[\dfrac{1}{x^2+1}\leq 1,\] jer je ta tvrdnja ekvivalentna tvrdnji \(x^2\geq 0\). Vrijedi i \(1\in S\), i to za \(x=0\). Dakle, \(\max{S}=1\), te je \(\max{S}=\sup{S}\).

Uvrštavanjem velikih brojeva umjesto \(x\) naslućujemo da je \(\inf{S}=0\). Pretpostavimo suprotno, da je \(a=\inf{S}\neq 0\). Kako je \(0\) jedna donja međa od \(S\), slijedi \(a>0\). Kako je \(a\) jedna donja međa od \(S\), vrijedi \(\dfrac{1}{x^2+1}\geq a\) za sve \(x\in \mathbb{R}\), odakle imamo
\[\dfrac{1}{x^2+1}\geq a\Leftrightarrow x^2+1\leq \dfrac{1}{a}\Leftrightarrow x^2\leq \dfrac{1}{a}-1,\]
što je u kontradikciji s činjenicom da je skup \(\{x^2 : x\in \mathbb{R}\}\) odozgo neograničen. Zaista, kad bi postojao $M>0$ takav da je $x^2\leq M$ za sve $x\in \mathbb{R}$, onda posebno za $x=M+1$ dobivamo $(M+1)^2\leq M$, no vrijedi $(M+1)^2=M^2+2M+1>2M\geq M$, što daje kontradikciju.

Nadalje, tvrdimo da \(\min{S}\) ne postoji. Pretpostavimo suprotno, da postoji \(\min{S}\). Tada je \(\min{S}=\inf{S}=0\). No ovo je nemoguće, jer vrijedi \(0\notin S\). Zaista, pretpostavimo da je \(0\in S\). Tada postoji \(x\in \mathbb{R}\) takav da je \(0=\dfrac{1}{x^2+1}\). Kako vrijedi \(\dfrac{1}{x^2+1}>0\), dobivamo kontradikciju.
\end{proof}

\begin{exercise}
Neka je $S=\left\{\dfrac{1}{x+1} : x\geq 0\right\}$. Odredite \(\inf{S}, \sup{S}, \min{S}, \max{S}\), ako postoje.
\end{exercise}
\begin{proof}[Rješenje]
Vrijedi $\sup{S}=\max{S}=1$, jer za sve $x\geq 0$ vrijedi $\dfrac{1}{x+1}\leq 1$ i $1\in S$, za $x=0$.

Tvrdimo da je $\inf{S}=0$. Očito je $\dfrac{1}{x+1}\geq 0$ za sve $x\geq 0$. Ovo se kao i prije može dokazati kontradikcijom, ali primijenimo ovaj put lemu \ref{26}, tj. dokažimo da za proizvoljan \(\epsilon>0\) postoji \(a\in S\) takav da je \(\epsilon>a\). To je ekvivalentno tvrdnji da za proizvoljan \(\epsilon>0\) postoji \(x\geq 0\) takav da je \[\dfrac{1}{x+1}<\epsilon.\] Da bismo dobili ideju kako bi \(x\) trebao izgledati, riješimo jednadžbu \(\dfrac{1}{x+1}=\epsilon\) po \(x\). Imamo
\[\dfrac{1}{x+1}=\epsilon \Leftrightarrow (x+1)\epsilon=1\Leftrightarrow x\epsilon=1-\epsilon\Leftrightarrow x=\dfrac{1-\epsilon}{\epsilon}=\dfrac{1}{\epsilon}-1.\]
Kako je \(\dfrac{1}{\epsilon}-1\) rješenje jednadžbe \(\dfrac{1}{x+1}=\epsilon\), slijedi da je \[\dfrac{1}{\left(\dfrac{1}{\epsilon}-1\right)+1}=\epsilon.\] Sada trebamo taj izbor \(x\)-a "popraviti" tako da za novi izbor \(x\)-a vrijedi \(\dfrac{1}{x+1}<\epsilon\). Kako je \[\dfrac{1}{\left(\dfrac{1}{\epsilon}-1\right)+1}=\epsilon>0,\] očito ako mu "povećamo" nazivnik, broj kojim time dobivamo bit će sigurno manji od \(\epsilon\). Zato ima smisla uzeti da je \[x=\dfrac{1}{\epsilon}+1.\] Tada vrijedi
\[x>\dfrac{1}{\epsilon}-1\Rightarrow x+1>\dfrac{1}{\epsilon}\Rightarrow \dfrac{1}{x+1}<\dfrac{1}{\dfrac{1}{\epsilon}}=\epsilon,\]
čime smo dokazali tvrdnju.

Nadalje, $\min{S}$ ne postoji, jer jednadžba $\dfrac{1}{x+1}=0$ nema rješenja.
\end{proof}

\[1-\dfrac{10}{x^2+5}>1-\epsilon,\] 
što je ekvivalentno uvjetu \[\dfrac{10}{x^2+5}<\epsilon.\]
Ovdje ćemo primijeniti sljedeći "trik" -- pronaći ćemo neki izraz koji je veći od $\dfrac{10}{x^2+5}$, ali koji i dalje možemo napraviti "po volji malim" i to na jednostavniji način. Primijetimo da je \[\dfrac{10}{x^2+5}<\dfrac{10}{x}\] za sve \(x>0\). Zaista vrijedi
$$\dfrac{10}{x^2+5}<\dfrac{10}{x}\Leftrightarrow x<x^2+5\Leftrightarrow x^2-x+5>0,$$
a posljednje vrijedi ne samo za sve $x>0$, nego i za sve $x\in \mathbb{R}$, budući da je vodeći koeficijent pozitivan i diskriminanta negativna.

\end{proof}

Kroz rješavanje prethodnih zadataka pokazalo se je da je rješavanje jednadžbe po \(x\) često od pomoći, pa rješavanjem jednadžbe \(\dfrac{10}{x^2+5}=\epsilon\) dobivamo \[x=\sqrt{\dfrac{10}{\epsilon}-5}=\sqrt{\dfrac{10-5\epsilon}{\epsilon}}.\] No primijetimo da ovaj izraz nije definiran ako je \(10-5\epsilon<0\), tj. ako je \(\epsilon>2\). \end{remark}

\begin{exercise}
\label{unbounded}
\item[a)] Neka je \(S=\left\{\dfrac{5}{x+2} : x\in \langle -2,0]\right\}\). Odredite \(\min{S}\). Odredite \(\inf{S}, \sup{S}, \min{S}, \max{S}\), ako postoje.
\item[b)] Neka je \(A=\left\{\dfrac{4x}{4x^2-1} : x>\dfrac{1}{2}\right\}\). Odredite \(\inf{S}, \sup{S}, \min{S}, \max{S}\), ako postoje.
\end{exercise}
\begin{proof}[Rješenje]
a) Za sve $x\in \langle -2, 0]$ vrijedi $\dfrac{5}{x+2}\geq \dfrac{5}{2}$ i ta vrijednost se postiže za $x=0$, pa je $\inf{S}=\min{S}=\dfrac{5}{2}$.

Tvrdimo da je $S$ odozgo neograničen. Zaista, pretpostavimo da postoji $M\in \mathbb{R}$ takav da za svaki $x\in \langle -2, 0]$ vrijedi $\dfrac{5}{x+2}\leq M$. Uočimo da rješavanjem jednadžbe $\dfrac{5}{x+2}=M$ dobivamo $x=\dfrac{5}{M}-2$, pa bi trebali dobiti kontradikciju ako "smanjimo" $x$, stoga ima smisla promatrati npr. $$x_0=\dfrac{5}{M+1}-2.$$ Uočimo da je $-2<x_0\leq 0$. Zaista, vrijedi $\dfrac{5}{M}-2\leq 0$ ako i samo ako je $M\geq \dfrac{5}{2}$, a to dobivamo iz pretpostavke za $x=0$, te vrijedi $-2<\dfrac{5}{M}-2$ budući da je očito $M>0$. Nadalje,
$$\dfrac{5}{x_0+2}=\dfrac{5}{\dfrac{5}{M+1}}=M+1>M,$$
što daje kontradikciju.

b) Tvrdimo da je $S$ odozgo neograničen. Ideja će biti pronaći neki izraz koji je manji od $\dfrac{4x}{4x^2-1}$, ali da i dalje teži u beskonačnost kako se približavamo $\dfrac{1}{2}$. Zaista, za sve $x\in \left\langle \dfrac{1}{2}, \infty\right\rangle$ vrijedi
$$\dfrac{4x}{4x^2-1}=\dfrac{4x}{(2x-1)(2x+1)}> \dfrac{4x}{(2x-1)\cdot 4x}=\dfrac{1}{2x+1},$$
jer je
$$\dfrac{4x}{(2x-1)(2x+1)}>\dfrac{4x}{(2x-1)4x}\Leftrightarrow 4x>2x+1\Leftrightarrow x>\dfrac{1}{2}.$$
Pretpostavimo sada da postoji $M\in \mathbb{R}$ takav da je $\dfrac{4x}{4x^2-1}\leq M$ za sve $x>\dfrac{1}{2}$. Očito je $M>0$, pa uzmemo li $x_0:=\dfrac{M+1}{2}$, vrijedi $x_0>\dfrac{1}{2}$ i $\dfrac{4x_0}{4x_0^2-1}>\dfrac{1}{2x_0-1}=M$, što daje kontradikciju.

Tvrdimo da je $\inf{S}=0$.

\textit{Prvi način.} Pretpostavimo da $a=\inf{S}\neq 0$. Kako je $0$ donja međa, slijedi da je $a>0$, te za sve $x>\dfrac{1}{2}$ vrijedi $\dfrac{4x}{4x^2-1}\geq a$, te
$$\dfrac{4x}{4x^2-1}\geq a \Leftrightarrow \dfrac{4x^2-1}{4x}\leq \dfrac{1}{a}\Leftrightarrow x-\dfrac{1}{4x}\leq \dfrac{1}{a}.$$
Preostaje dokazati da je skup
$$A:\left\{x-\dfrac{1}{4x} : x>\dfrac{1}{2}\right\}$$
odozgo neograničen. Uočimo da za sve $x>\dfrac{1}{2}$ vrijedi $x-\dfrac{1}{4x}>x-\dfrac{1}{2}$, jer je
$$x-\dfrac{1}{4x}>x-\dfrac{1}{2}\Leftrightarrow \dfrac{1}{4x}<\dfrac{1}{2}\Leftrightarrow x>\dfrac{1}{2}.$$
Kao i prije, pretpostavimo li da postoji $M\in \mathbb{R}$ takav da je $x-\dfrac{1}{4x}\leq M$ za sve $x>\dfrac{1}{2}$. Kako je $M>0$, možemo uzeti $x_0:=M+\dfrac{1}{2}$. Vrijedi $x_0-\dfrac{1}{4x_0}>x_0-\dfrac{1}{2}=M$, čime dobivamo kontradikciju.

pa rješavanjem jednadžbe $\dfrac{2}{2x_0-1}=\epsilon$ dobivamo $x_0=\dfrac{1}{\epsilon}+\dfrac{1}{2}$, što je upravo traženi $x_0$.

\begin{lemma}
Neka su \(A, B\subseteq \mathbb{R}\) neprazni, \(A\subseteq B\) i neka je \(B\) odozgo (odozdo) ograničen skup. Tada je i \(A\) odozgo (odozdo) ograničen skup, te vrijedi \(\sup{A}\leq \sup{B}\) (\(\inf{A}\geq \inf{B}\)).
\end{lemma}
Uočimo za prethodna lema povlači sljedeću tvrdnju -- Neka je \(A\subseteq \mathbb{R}\) neprazan i odozdo ograničen, te neka je \(a\in \mathbb{R}\) donja međa skupa \(A\). Ako postoji \(B\subseteq A\) sa svojstvom da je \(\inf{B}=a\), onda je \(a=\inf{A}\). Zaista, iz prethodne leme slijedi da je \(B\) odozdo ograničen, te \(a=\inf{B}\geq \inf{A}\). S druge strane, po definiciji infimuma vrijedi \(a\leq \inf{A}\), pa je zaista \(a=\inf{A}\). Analogna tvrdnja vrijedi za odozgo ograničene skupove i supremum.

\begin{remark}

\begin{itemize}
\item[a)] Ako su \(A\), \(B\subseteq \mathbb{R}\) odozgo (odozdo) ograničeni skupovi, onda je skup
\[A+B=\{a+b : a\in A,\; b\in B\}\]
odozgo (odozdo) ograničen i vrijedi
\[\sup(A+B)=\sup{A}+\sup{B}\hspace{0.5cm}\left(\inf(A+B)=\inf{A}+\inf{B}\right).\]
\item[b)] Ako su \(A\), \(B\in \mathbb{R}\) odozgo ograničeni skupovi, takvi da je \(a\geq 0\) za svaki \(a\in A\), te \(b\geq 0\) za svaki \(b\in B\) (kraće: \(A\geq 0\) i \(B\geq 0\), dakle oni su i odozdo ograničeni), tada je skup
\[A\cdot B=\{a\cdot b : a\in A,\; b\in B\}\]
ograničen i vrijedi
\[\sup(A\cdot B)=\sup{A}\cdot \sup{B} \; \wedge \; \inf(A\cdot B)=\inf{A}\cdot \inf{B}.\]
\item[c)] Ako je \(A\subseteq \mathbb{R}\) odozdo (odozgo) ograničen skup, tada je skup
\[-A=\{-a : a\in A\}\]
odozgo (odozdo) ograničen i vrijedi
\[\sup(-A)=-\inf{A}, \hspace{0.5cm} \left(\inf(-A)=-\sup{A}\right).\]
\item[d)] Ako su \(A\), \(B\in \mathbb{R}\) odozgo (odozdo) ograničeni skupovi, tada je skup \(A\cup B\) odozgo (odozdo) ograničen i vrijedi
\[\sup(A\cup B)=\max\{\sup{A},\sup{B}\},\hspace{0.5cm} \left(\inf(A\cup B)=\min\{\inf{A}, \inf{B}\}\right).\]
\end{itemize}
\end{remark}

\begin{remark}
\label{unprfam}
Neka su \(S,I\) neprazni skupovi i \(\mathcal{F}=\left\{A_n : n\in I\right\}\) familija skupova (skup nekih podskupova od \(S\)), dakle \(A_n\subseteq S\), za sve \(n\in I\). Definiramo:

\label{pr}
\bigcap_{n\in I}{A_n}:=\left\{x\in S : (\forall n\in I)\; x\in A_n\right\}.
\end{gather}
Skup (\ref{un}) zovemo \textbf{unija familije \(\mathcal{F}\)}, (\ref{pr}) zovemo \textbf{presjek familije \(\mathcal{F}\)}. Unije i presjeci familija se inače definiraju i za neindeksirane familije, ali ovo će za nas biti dovoljno. Više o familijama skupova možete vidjeti u \cite{9}, str. 9.
\end{remark}

\begin{exercise}[Cantorov aksiom]
\label{Cantor}
Neka je za sve \(n\in \mathbb{N}\) zadan segment \([a_n, b_n]\subseteq \mathbb{R}\) i neka za sve \(m\in \mathbb{N}\), \(m\geq n\) povlači \([a_m, b_m]\subseteq [a_n, b_n]\). Dokažite:
\[\bigcap_{n\in \mathbb{N}}{[a_n, b_n]}\neq \emptyset.\]
\end{exercise}
\begin{proof}[Rješenje]
Naš je cilj dokazati da postoji \(b\in \mathbb{R}\) takav da za sve \(n\in \mathbb{N}\) vrijedi \(a_n\leq b\leq b_n\). Uočimo prvo da je tvrdnja \([a_m, b_m]\subseteq [a_n, b_n]\) ekvivalentna tvrdnji \(a_n\leq a_m\leq b_m\leq b_n\). Neka je sada \[A=\{a_n : n\in \mathbb{N}\}\;\;\text{i}\;\;B=\{b_n : n\in \mathbb{N}\}.\] Uočimo da za sve \(n_1, n_2\in \mathbb{N}\) vrijedi \(a_{n_1}\leq b_{n_2}\). Zaista, ako je \(n_1\leq n_2\), onda vrijedi \(a_{n_1}\leq a_{n_2}\leq b_{n_2}\), a ako je \(n_2\leq n_1\), onda je \(a_{n_1}\leq b_{n+1}\leq b_{n_2}\). Kako je svaki element skupa \(A\) jedna donja međa od \(B\), očito postoji \(b=\inf{B}\). Očito je \(b_n\geq b\) za sve \(n\in \mathbb{N}\). Tvrdimo da za sve \(n\in \mathbb{N}\) vrijedi \(a_n\leq b\). Pretpostavimo suprotno, tj. da postoji \(n_1\in \mathbb{N}\) takav da je \(b<a_{n_1}\). Tada \(a_{n_1}\) sigurno nije donja međa skupa \(B\), pa postoji \(n_2\in \mathbb{N}\) takav da je \(b_{n_2}<a_{n_1}\), kontradikcija s \(A\leq B\)! Dakle, vrijedi \(a_n\leq b\leq b_n\), pa je tvrdnja dokazana.
\end{proof}
\begin{definition}
Za funkciju \(p : \mathbb{R}^2\to \mathbb{R}\) kažemo da je \textbf{polinom \(n+m\)-tog stupnja u dvije varijable} (kraće: polinom u dvije varijable) ako postoje brojevi \(i=1, \dots, n\), \(j=1,\dots, m\) i \(a_{ij}\in \mathbb{R}\) takvi da je
\[p(x, y)=\sum_{i=0}^n{\sum_{j=0}^m{{a_{ij}}x^{i}y^j}},\]
pri čemu nisu svi \(a_{ij}\) takvi da je \(i+j=n+m\) jednaki \(0\).
\end{definition}
Tako su npr. \(p_1, p_2 : \mathbb{R}^2\to \mathbb{R}\),
\begin{gather*}
p_1(x, y)=x^2+2xy+3y^2,\\
p_2(x, y)=xy+2
\end{gather*}
dva polinoma u dvije varijable.
\begin{exercise}
Dokažite da postoji polinom u dvije varijable \(p : \mathbb{R}^2\to \mathbb{R}\) takav da je \(p(x, y)>0\) za sve \((x, y)\in \mathbb{R}^2\), koji nema minimum, tj. ne postoji \(\min\{p(x, y) : x\in \mathbb{R},\; y\in \mathbb{R}\}.\)
\end{exercise}
\begin{proof}[Rješenje]
Označimo \(S=\{p(x, y) : x\in \mathbb{R},\; y\in \mathbb{R}\}\). Definirat ćemo polinom drugog stupnja za koji je \(\inf{S}=0\), ali \(0\notin S\). Korisna opservacija je da za polinome \(q : \mathbb{R}^2\to \mathbb{R}\) zapisane u obliku
\[q(x, y)=(\dots)^2+(\dots)^2\]
vrijedi \(q(x, y)>0\) ako (i samo ako) izrazi u ove dvije zagrade ne mogu istodobno biti \(0\). Dakle, nama je cilj u ove dvije zagrade dodati izraze (najprirodnije je dodati polinome) koji ne mogu istodobno biti \(0\), ali tako da infimum skupa \(S\) bude jednak \(0\), tj. tako da možemo odabrati brojeve \(x, y\) tako da izraz \(p(x, y)\) bude "proizvoljno malen". Možda je najjednostavnije u prvu zagradu staviti izraz \(x\). U drugu zagradu ćemo morati staviti izraz koji ne može biti \(0\) kada je \(x=0\). To možemo postići tako da npr. uzmemo neki izraz koji je \(0\) kada je \(x=0\) (možemo naprosto uzeti neki izraz koji sadrži u sebi \(x\) kao faktor) i tom izrazu dodamo \(1\). Tu je najlakše uzeti \(xy+1\) kao taj izraz u drugoj zagradi\footnote{Uočimo da npr. \(x^2+1\) ne bi bio dobar odabir, jer ne bi izraze u obije zagrade istovremeno mogli učiniti po volji malima.}, jer taj izraz možemo skupa s izrazom u prvoj zagradi učiniti po volji malim (uzimajući \(x\) vrlo blizu \(0\) da prva zagrada bude mali broj i kao \(y\) suprotnu i recipročnu vrijednost tog broja kojeg smo uzeli za \(x\), kako bi druga zagrada bila \(0\)). Dakle, uzet ćemo
\[p(x, y)=x^2+(xy+1)^2.\]
Sad ćemo formalizirati naša razmatranja. Tvrdimo da za skup
\[S=\{x^2+(xy+1)^2 : x\in \mathbb{R},\; y\in \mathbb{R}\}\]
vrijedi \(\inf{S}=0\), te \(0\notin S\). Uočimo da je očito \(x^2+(xy+1)^2\geq 0\) i vrijedi \(x^2+(xy+1)^2=0\) ako i samo ako je \(x=0\) i \(xy=-1\), što je nemoguće, dakle vrijedi \(x^2+(xy+1)^2>0\) za sve \(x, y\in \mathbb{R}\), što povlači \(0\notin S\).

Dokažimo da je \(\inf{S}=0\). Treba pokazati da za proizvoljan \(\epsilon>0\) postoje \(x, y\in \mathbb{R}\) tako da vrijedi
\[\epsilon> x^2+(xy+1)^2\]
Uzmimo \(x=\dfrac{\sqrt{\epsilon}}{2}\) i \(y=-\dfrac{2}{\sqrt{\epsilon}}\). Tada je
\[x^2+(xy+1)^2=\dfrac{\epsilon}{4}<\epsilon.\]
Dakle, \(p : \mathbb{R}^2\to \mathbb{R}\), \(p(x, y)=x^2+(xy+1)^2\) je jedan polinom koji zadovoljava uvjete zadatka.
\end{proof}
U idućem zadatku zanimat će nas problem egzistencije $n$-tog korijena. Naime, iz aksioma realnih brojeva nije apriori jasno da je $n$-ti korijen nešto što je dobro definirano.
\begin{exercise}
Dokažite da za svaki $x>0$ i svaki $n\in \mathbb{N}$ postoji točno jedan $y>0$ takav da je $y^n=x$.
\end{exercise}
\begin{proof}[Rješenje]
\textit{Jedinstvenost.} Kad bi postojali $0<y_1<y_2$ takvi da je $y_1^n=y_2^n=x$, imali bi ujedno i $y_1^n<y_2^n$, što je nemoguće.

\textit{Egzistencija.} Ideja će biti dobiti traženi $y$ tako da ga prikažemo kao supremum nekog skupa. Zaista, neka su $x>0$ i $n\in \mathbb{N}$, $n\geq 2$ proizvoljni (za $n=1$ tvrdnja je trivijalna), te promotrimo skup
$$E:=\left\{ t>0 : t^n<x\right\}.$$
Najprije uočimo da je ovaj skup neprazan. Zaista, ako je $x\in \langle 0, 1\rangle$, onda je $x^n<x$, pa je $x\in E$, a ako je $x\geq 1$, možemo uzeti $t=\dfrac{1}{2}$, pa je $t^n<t< 1\leq x$. Nadalje, $E$ je odozgo ograničen i to s $M=\max\left\{1, x\right\}$. Zaista, neka je $t\in E$. Ako je $t\leq 1$ onda je očito $t\leq M$, a ako je $t>1$, onda je $t< t^n<x\leq M$. Stoga $E$ ima supremum, nazovimo ga s $y$. Tvrdimo da je $y^n=x$.

Pretpostavimo da je $y^n<x$. Tada tvrdimo da postoji neki "dovoljno mali" $h>0$ takav da je $y+h\in E$, tj. $(y+h)^n<x$. Uočimo da ukoliko je $0<h<1$ vrijedi
\begin{align*}
(y+h)^n&=y^n+\dbinom{n}{1}y^{n-1}h+\dbinom{n}{2}y^{n-2}h^2+...+h^n\\
&=y^n+h\left(\dbinom{n}{1}y^{n-1}+\dbinom{n}{2}y^{n-2}h+...+h^{n-1}\right)\\
&\leq y^n+h\left(\dbinom{n}{1}y^{n-1}+\dbinom{n}{2}y^{n-2}+...+1\right).
\end{align*}
Sada ako uzmemo
$h:=\min\left\{1, \dfrac{x-y^n}{\dbinom{n}{1}y^{n-1}+\dbinom{n}{2}y^{n-2}+...+1}\right\}$,
dobivamo da je $(y+h)^n<x$, što je i trebalo pokazati.

Pretpostavimo sada da je $y^n>x$. Tvrdimo da tada postoji "dovoljno mali" $h>0$ takav da je $y-h$ gornja međa skupa $E$. Drugim riječima, tvrdimo da vrijedi $t\leq y-h$ za sve $t\in E$. Za to je dovoljno pokazati da je $x<(y-h)^n$ za "dovoljno mali" $h$, jer odatle slijedi $t^n<x<(y-h)^n$, tj. $t<y-h$. Vrijedi
\begin{align*}
(y-h)^n&=y^n-\dbinom{n}{1}y^{n-1}h+\dbinom{n}{2}y^{n-2}h^2+...+(-1)^{n+1}h^n\\
&>y^n-\dbinom{n}{1}y^{n-1}h-\dbinom{n}{3}y^{n-3}h^3-...,
\end{align*}
gdje smo u posljednjoj nejednakosti iz $-\dbinom{n}{1}y^{n-1}h+\dbinom{n}{2}y^{n-2}h^2+...+(-1)^{n+1}h^n$ "izbacili" sve pozitivne članove. Dakle, za
$h:=\min\left\{1, \dfrac{y^n-x}{\dbinom{n}{1}y^{n-1}+\dbinom{n}{3}y^{n-3}+...}\right\}$,
vrijedi
\begin{align*}
y^n-\dbinom{n}{1}y^{n-1}h-\dbinom{n}{3}y^{n-3}h^3-...&=y^n-h\left(\dbinom{n}{1}y^{n-1}+\dbinom{n}{3}y^{n-3}h^2+...\right)\\
&>y^n-h\left(\dbinom{n}{1}y^{n-1}+\dbinom{n}{3}y^{n-3}+...\right)\\
&>y^n-\left(y^n-x\right)=x,
\end{align*}
što je i trebalo pokazati. Dakle $y-h$ je gornja međa, što je u suprotnosti s definicijom supremuma.
\end{proof}

\begin{definition}
Skup kompleksnih brojeva je skup \(\mathbb{C}=\left\{a+bi : a,\; b\in \mathbb{R},\, i=\sqrt{-1}\right\}\).
(Ova definicija nije stroga, strogu definiciju obradili ste na predavanjima.)

\noindent Kompleksne brojeve zbrajamo i oduzimamo "po komponentama", tj. \[(a+bi)\pm (c+di)=(a+c)\pm (b+d)i,\] te ih množimo poput polinoma, tj. \[(a+bi)(c+di)=ac+adi+bci-bd=(ac-bd)+(ad+bc)i.\] Vrijedi i \[\dfrac{1}{a+bi}=\dfrac{a-bi}{a^2+b^2},\] što se lako pamti kao "racionalizacija nazivnika", tj. množimo brojnik i nazivnik s \(a-bi\), pa u nazivniku dobivamo \[(a+bi)(a-bi)=a^2-abi+abi+b^2=a^2+b^2.\]

\begin{gather*}
z_1z_2=r_1r_2\left(\cos(\theta_1+\theta_2)+i\sin(\theta_1+\theta_2)\right),\\
\dfrac{z_1}{z_2}=\dfrac{r_1}{r_2}\left(\cos(\theta_1-\theta_2)+i\sin(\theta_1-\theta_2)\right), \;\mathrm{gdje}\;\mathrm{je}\; z_2\neq 0.
\end{gather*}
\item Za sve \(z\in \mathbb{C}\) i \(n\in \mathbb{N}\) vrijedi \(z^n=r^n(\cos{n\theta}+i\sin{n\theta})\).
\end{itemize}
\end{remark}

\begin{exercise} \textbf{}
\begin{itemize}
\item[a)] Zapišite \(z=1-\sqrt{3}i\) u trigonometrijskom obliku. Odredite \(z^5\).
\item[b)] Odredite najmanji \(\alpha\in \mathbb{N}\) takav da je \(z=\left(1+i\right)^{\alpha}\left(-\dfrac{1}{2}+\dfrac{\sqrt{3}}{2}i\right)^{\alpha}\in \mathbb{R}\).
\end{itemize}
\end{exercise}
\begin{proof}[Rješenje]
a) \(z=2\left(\cos\left(-\dfrac{\pi}{3}\right)+i\sin\left(-\dfrac{\pi}{3}\right)\right)\), \(z^5=32\left(\cos{\dfrac{\pi}{3}}+i\sin{\dfrac{\pi}{3}}\right)\).

b) Vrijedi 
\begin{align*}
\left(1+i\right)^{\alpha}\left(-\dfrac{1}{2}+\dfrac{\sqrt{3}}{2}i\right)^{\alpha}&=\sqrt{2}^{\alpha}\left(\cos{\dfrac{\alpha \pi}{4}}+i\sin{\dfrac{\alpha \pi}{4}}\right)\left(\cos{\dfrac{2\alpha \pi}{3}}+i\sin{\dfrac{2\alpha \pi}{3}}\right)\\
&=\sqrt{2}^{\alpha}\left(\cos{\dfrac{11\alpha \pi}{12}}+i\sin{\dfrac{11\alpha \pi}{12}}\right)
\end{align*}
Kompleksan broj \(z\) je realan ako i samo ako je \(\mathrm{Im}(z)=0\). Zato je najmanji takav \(\alpha\in \mathbb{N}\) upravo najmanji \(\alpha\) koji zadovoljava \(\sin{\dfrac{11\alpha \pi}{12}}=0\), a to je \(\alpha=12\).
\end{proof}
